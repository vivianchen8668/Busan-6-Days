function toggleDetail(index) {

const detail = document.getElementById(`detail-${index}`);

const icon = document.getElementById(`icon-${index}`);

if(detail.classList.contains('hidden')) {

detail.classList.remove('hidden');

icon.classList.replace('fa-chevron-down', 'fa-chevron-up');

} else {

detail.classList.add('hidden');

icon.classList.replace('fa-chevron-up', 'fa-chevron-down');

}

}

 

// 個人記帳匯率計算

function calcPersonalCurrency() {

const amt = parseFloat(document.getElementById('exp-amount').value) || 0;

const curr = document.getElementById('exp-currency').value;

const convertedEl = document.getElementById('exp-converted');

 

if(curr === 'KRW') {

const twd = Math.round(amt * KRW_RATE);

convertedEl.innerText = `換算台幣：NT$ ${twd}`;

} else {

const krw = Math.round(amt / KRW_RATE);

convertedEl.innerText = `換算韓幣：₩ ${krw}`;

}

}

 

function addPersonalExpense() {

const title = document.getElementById('exp-title').value.trim();

const amt = parseFloat(document.getElementById('exp-amount').value) || 0;

const curr = document.getElementById('exp-currency').value;

 

if(!title || amt <= 0) return alert('請填寫完整項目與金額！');

 

const twdAmt = curr === 'KRW' ? Math.round(amt * KRW_RATE) : amt;

personalExpenses.push({ title, amt, curr, twdAmt });

 

// 清空輸入

document.getElementById('exp-title').value = '';

document.getElementById('exp-amount').value = '';

calcPersonalCurrency();

 

renderPersonalExpenses();

}

 

function renderPersonalExpenses() {

const list = document.getElementById('personal-expense-list');

if(personalExpenses.length === 0) {

list.innerHTML = `<p class="text-xs text-gray-400 text-center py-4">尚無記帳紀錄</p>`;

return;

}

list.innerHTML = '';

personalExpenses.forEach((item, idx) => {

list.insertAdjacentHTML('beforeend', `

<div class="flex justify-between items-center text-xs p-2 bg-gray-50 rounded-xl">

<span class="font-medium text-gray-700">${item.title}</span>

<span class="font-bold text-[#8c7355]">${item.curr === 'KRW' ? '₩ ' + item.amt : 'NT$ ' + item.amt} <span class="text-[10px] text-gray-400 font-normal">(NT$ ${item.twdAmt})</span></span>

</div>

`);

});

}
